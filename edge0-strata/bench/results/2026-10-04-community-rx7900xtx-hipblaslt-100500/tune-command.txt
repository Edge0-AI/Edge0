# Reproduce the gfx1100 hipBLASLt 100500 tuning table

# Tuner binary: preserved from the 0.1.38-lineage build (same tree as the nightly
# candidate; rebuildable from the fork source with `cmake --build <bld> --target tune_hipblaslt`).
# It links hipBLASLt dynamically, so pointing
# LD_LIBRARY_PATH at the ROCm 10.2 nightly SDK's lib makes it tune and stamp
# the table for THAT library (hipblasLtGetVersion() is queried at run time).
SDK=/home/zacch/strata-rocm-experiments/20261003-231142/sdk
LD_LIBRARY_PATH=$SDK/lib /home/zacch/strata-rocm-experiments/20261003-231142/tune_hipblaslt \
  --shapes-file shapes-canon.json \
  --tokens 4096,8192 \
  --workspace-mib 32 \
  --tuning-out gfx1100-hipblaslt-100500.txt

# meta line to expect: device_arch=gfx1100 hipblaslt_version=100500
# 32 cases (16 shapes x {4096,8192} token buckets), every candidate
# correctness-gated by the tuner (rel-L2 1e-4 / max-abs 1e-2, padding canary);
# each table row is its case's measured best.  ~1 GB VRAM, a few minutes.
