# Minimal host project for subproject_test.py: adds the engine with add_subdirectory() and records
# what leaked into the host. It sets no C++ standard of its own on purpose.
cmake_minimum_required(VERSION 3.21)
project(edge0_subproject_probe LANGUAGES C CXX)
set(EDGE0_SOURCE_DIR "" CACHE PATH "Engine source tree")
set(PROBE_OUTPUT "" CACHE FILEPATH "File the probes are written to")
add_subdirectory("${EDGE0_SOURCE_DIR}" edge0)

# A host library without an explicit type must stay static (BUILD_SHARED_LIBS must not leak).
add_library(probe_host_lib host_lib.cpp)
# Compiled with C++11 requested: edge0::edge0 must raise it to C++17 for the public headers.
add_library(probe_consumer OBJECT main.cpp)
set_target_properties(probe_consumer PROPERTIES CXX_STANDARD 11 CXX_EXTENSIONS OFF)
target_link_libraries(probe_consumer PRIVATE edge0::edge0)

set(lines "")
foreach(target IN ITEMS edge0-infer moe_example_native_chat moe_test_app moe_public_header_check)
    if(TARGET ${target})
        list(APPEND lines "target.${target}=present")
    else()
        list(APPEND lines "target.${target}=absent")
    endif()
endforeach()
get_target_property(host_lib_type probe_host_lib TYPE)
list(APPEND lines "host_lib_type=${host_lib_type}")
if(DEFINED CACHE{BUILD_SHARED_LIBS})
    list(APPEND lines "cache.BUILD_SHARED_LIBS=$CACHE{BUILD_SHARED_LIBS}")
else()
    list(APPEND lines "cache.BUILD_SHARED_LIBS=<unset>")
endif()
get_directory_property(engine_output DIRECTORY "${EDGE0_SOURCE_DIR}" DEFINITION CMAKE_RUNTIME_OUTPUT_DIRECTORY)
list(APPEND lines "engine.CMAKE_RUNTIME_OUTPUT_DIRECTORY=${engine_output}")
get_directory_property(identity DIRECTORY "${EDGE0_SOURCE_DIR}" DEFINITION MOE_BUILD_IDENTITY_HEADER)
list(APPEND lines "engine.MOE_BUILD_IDENTITY_HEADER=${identity}")
list(APPEND lines "engine.binary_dir=${CMAKE_BINARY_DIR}/edge0")
foreach(option IN ITEMS GGML_BLAS GGML_ACCELERATE GGML_LLAMAFILE GGML_OPENMP GGML_METAL GGML_NATIVE GGML_CPU_ARM_ARCH)
    list(APPEND lines "cache.${option}=$CACHE{${option}}")
endforeach()
list(JOIN lines "\n" text)
file(WRITE "${PROBE_OUTPUT}" "${text}\n")
